CREATE TRIGGER [dbo].[trg_CapNhatSoLuongGhe]
ON [dbo].[Ghe]
AFTER INSERT, DELETE
AS
BEGIN
    UPDATE PhongChieu
    SET PC_SLGhe = (SELECT COUNT(*) FROM Ghe WHERE Ghe.PC_Ma = PhongChieu.PC_Ma)
    WHERE PC_Ma IN (
        SELECT DISTINCT PC_Ma FROM inserted
        UNION
        SELECT DISTINCT PC_Ma FROM deleted
    );
END;
GO

CREATE TRIGGER [dbo].[TR_SuatChieu_KiemTraLich]
ON [dbo].[SuatChieu]
AFTER INSERT, UPDATE
AS
BEGIN
    SET NOCOUNT ON;

    IF EXISTS
    (
        SELECT 1
        FROM inserted I
        JOIN SuatChieu SC
            ON I.PC_Ma = SC.PC_Ma
            AND I.SC_NgayChieu = SC.SC_NgayChieu
        WHERE I.SC_Ma <> SC.SC_Ma
          AND I.SC_GioBatDau < SC.SC_GioKetThuc
          AND I.SC_GioKetThuc > SC.SC_GioBatDau
    )
    OR EXISTS
    (
        SELECT 1
        FROM inserted I
        JOIN PhongChieu PC ON I.PC_Ma = PC.PC_Ma
        WHERE PC.PC_TrangThai <> N'Hoạt động'
    )
    BEGIN
        RAISERROR(
            N'Lịch chiếu bị trùng giờ hoặc phòng chiếu không hoạt động!',
            16, 1
        );

        ROLLBACK TRANSACTION;
        RETURN;
    END
END;

GO
ALTER TABLE [dbo].[SuatChieu] ENABLE TRIGGER [TR_SuatChieu_KiemTraLich]
GO

        FROM inserted I

        JOIN Ghe G ON I.G_Ma = G.G_Ma

        );

        ROLLBACK TRANSACTION;
        RETURN;
    END
